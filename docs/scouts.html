not_trained_df <- scouts_df |>
  dplyr::filter(Trained == "NO") |>
  dplyr::arrange(MemberID) |>
  dplyr::select(-c("Council", "Service_Area", "Sub_District", "Gender_Accepted", "Chartered_Org_Name", "Zip_Code"))

not_trained_df |>

How many positions per person?

```{r}
not_trained_count_df <- not_trained_df |>
  dplyr::group_by(MemberID, Last_Name, First_Name, Email) |>
  dplyr::summarise(n_pos = dplyr::n()) |>
  dplyr::arrange(desc(n_pos)) 

readr::write_csv(not_trained_count_df, file = "include/csv/not-trained-count.csv")
